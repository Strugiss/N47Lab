def _generate_blender_donut(R, r, major_segs, minor_segs):
    """
    Genera una ciambella (toro) tramite implementazione parametrica vettorializzata.
    """
    major_segs = max(8, int(major_segs))
    minor_segs = max(4, int(minor_segs))
    
    if R <= r + 0.1:
        R = r + 0.2
        print(f"⚠️ Avviso: Raggio maggiore della ciambella impostato a {R:.2f} per garantire un foro centrale visibile.")
    
    # Generazione coordinate parametriche
    u = np.linspace(0, 2 * np.pi, major_segs, endpoint=False)
    v = np.linspace(0, 2 * np.pi, minor_segs, endpoint=False)
    U, V = np.meshgrid(u, v, indexing='ij')
    
    # Calcolo coordinate cartesiane
    X = (R + r * np.cos(V)) * np.cos(U)
    Y = (R + r * np.cos(V)) * np.sin(U)
    Z = r * np.sin(V)
    
    # Creazione vertici
    verts = np.stack([X, Y, Z], axis=-1).reshape(-1, 3)
    
    # Costruzione topologica ottimizzata
    faces = []
    for i in range(major_segs):
        i_next = (i + 1) % major_segs
        for j in range(minor_segs):
            j_next = (j + 1) % minor_segs
            
            v00 = i * minor_segs + j
            v10 = i_next * minor_segs + j
            v11 = i_next * minor_segs + j_next
            v01 = i * minor_segs + j_next
            
            faces.append([v00, v10, v11])
            faces.append([v00, v11, v01])
    
    mesh = trimesh.Trimesh(vertices=verts, faces=np.array(faces))
    mesh.fix_normals()
    
    return mesh

def _generate_blender_cylinder(radius, height, sections):
    mesh = trimesh.creation.cylinder(radius=radius, height=height, sections=int(sections))
    mesh.fix_normals()
    return mesh

def _generate_blender_sphere(radius, subdivisions):
    mesh = trimesh.creation.icosphere(radius=radius, subdivisions=min(6, int(subdivisions)))
    mesh.fix_normals()
    return mesh

def _generate_blender_cone(radius, height, sections):
    mesh = trimesh.creation.cone(radius=radius, height=height, sections=int(sections))
    mesh.fix_normals()
    return mesh

def _generate_blender_box(width, height, depth):
    mesh = trimesh.creation.box(extents=[width, height, depth])
    mesh.fix_normals()
    return mesh

def boolean_safe(mesh_a, mesh_b, operation):
    """
    Esegue operazioni di Geometria Solida Costruttiva (CSG) in modo sicuro.
    """
    op_map = {
        "unione": "union",
        "sottrazione": "difference",
        "intersezione": "intersection"
    }
    
    op_type = op_map.get(operation.lower(), operation.lower())
    
    try:
        return trimesh.boolean.boolean([mesh_a, mesh_b], operation=op_type)
    except Exception as e:
        raise RuntimeError(f"Errore motore CSG: {e}")

def validate_and_place_mesh(mesh):
    """Pipeline di validazione e posizionamento mesh."""
    try:
        if mesh is None or len(mesh.vertices) < 3 or len(mesh.faces) < 1:
            return _generate_blender_box(10, 10, 10)
        
        mesh.process(validate=True)
        mesh.fix_normals()
        
        if mesh.bounds is not None and len(mesh.bounds) == 2:
            z_min = mesh.bounds[0][2]
            if abs(z_min) > 1e-6:
                mesh.apply_translation([0, 0, -z_min])
        
        return mesh
    except Exception as e:
        print(f"⚠️ validate_and_place_mesh error: {e}")
        return _generate_blender_box(10, 10, 10)

def create_mesh(shape_type, params):
    try:
        if shape_type == "box":
            mesh = _generate_blender_box(
                params.get("larghezza", 20), 
                params.get("altezza", 20), 
                params.get("profondità", 20)
            )
        elif shape_type == "cylinder":
            mesh = _generate_blender_cylinder(
                params.get("raggio", 10), 
                params.get("altezza", 30), 
                int(params.get("sezioni", 64))
            )
        elif shape_type == "sphere":
            mesh = _generate_blender_sphere(
                params.get("raggio", 15), 
                int(params.get("suddivisioni", 4))
            )
        elif shape_type == "cone":
            mesh = _generate_blender_cone(
                params.get("raggio_base", 12), 
                params.get("altezza", 30), 
                int(params.get("sezioni", 64))
            )
        elif shape_type == "donut":
            R = params.get("raggio_magg", 15)
            r = params.get("raggio_min", 5)
            
            if R <= r + 0.1:
                R = r + 0.2
            
            mesh = _generate_blender_donut(
                R, 
                r, 
                int(params.get("sezioni", 64)), 
                int(params.get("sezioni", 64)) // 2
            )
        else:
            mesh = _generate_blender_box(10, 10, 10)
        
        return validate_and_place_mesh(mesh)
    except Exception as e:
        print(f"⚠️ create_mesh error: {e}")
        return validate_and_place_mesh(_generate_blender_box(10, 10, 10))

class Scene:
    """Classe centrale per la gestione della scena CAD."""
    
    def __init__(self):
        self.objects = []
        self.selected = None
        self.undo_stack = []
        self.redo_stack = []
        self.color_idx = 0
        self.sketch_entities = []
        self.dimensions = []
        self.angle_dims = []
        self.gcode_paths = []
        self.snap_grid = True
        self.scale_mode = "Disattivato"
        self.magnetic_snap = True
        self.layers = {"Default": {"visible": True, "locked": False, "color": [0.6, 0.75, 0.9, 1.0]}}
        self.active_layer = "Default"
        self.assemblies = {}
    
    @property
    def grid_step(self):
        return 0.0 if self.scale_mode == "Disattivato" else float(self.scale_mode.replace(" mm", ""))
    
    def add_shape(self, shape_type, params, name=None):
        mesh = create_mesh(shape_type, params)
        color = PASTEL_COLORS[self.color_idx % len(PASTEL_COLORS)]
        self.color_idx += 1
        
        mesh.metadata.update({
            "layer": self.active_layer,
            "color": color,
            "name": name or f"{shape_type}_{self.color_idx}",
            "params": params.copy(),
            "shape_type": shape_type,
            "assembly": None
        })
        
        self.objects.append(mesh)
        self._undo_push()
        return mesh
    
    def _undo_push(self):
        try:
            self.undo_stack.append({
                "obj": [obj.copy() for obj in self.objects if hasattr(obj, 'copy')]
            })
            if len(self.undo_stack) > 50:
                self.undo_stack.pop(0)
            self.redo_stack.clear()
        except Exception as e:
            print(f"⚠️ _undo_push error: {e}")
    
    def undo(self):
        if self.undo_stack:
            state = self.undo_stack.pop()
            self.redo_stack.append({"obj": [obj.copy() for obj in self.objects]})
            self.objects = state["obj"]
            self.selected = None
            self._notify("✅ Annullato")
            return True
        return False
    
    def redo(self):
        if self.redo_stack:
            state = self.redo_stack.pop()
            self.undo_stack.append({"obj": [obj.copy() for obj in self.objects]})
            self.objects = state["obj"]
            self.selected = None
            self._notify("✅ Ripristinato")
            return True
        return False
    
    def delete(self):
        if self.selected and self.selected in self.objects:
            self.objects.remove(self.selected)
            self.selected = None
            self._undo_push()
            self._notify("🗑️ Eliminato")
            return True
        return False
    
    def duplicate(self):
        if not self.selected:
            return self._notify("⚠️ Nessun oggetto selezionato")
        
        copy = self.selected.copy()
        copy.apply_translation([20, 0, 0])
        copy.metadata["name"] += "_copy"
        
        self.objects.append(copy)
        self.selected = copy
        self._undo_push()
        self._notify("📋 Duplicato")
    
    def align_z(self):
        if not self.selected:
            return self._notify("⚠️ Nessun oggetto selezionato")
        
        try:
            bounds = self.selected.bounds
            if bounds and len(bounds) == 2 and abs(bounds[0][2]) > 1e-3:
                self.selected.apply_translation([0, 0, -bounds[0][2]])
                self._undo_push()
                self._notify("📏 Allineato Z=0")
                return True
        except Exception as e:
            print(f"⚠️ align_z error: {e}")
        return False
    
    def ungroup_object(self):
        if self.selected and self.selected.metadata.get("assembly"):
            self.selected.metadata["assembly"] = None
            self._undo_push()
            self._notify("✂️ Separato")
    
    def explode_assembly(self, assembly_name):
        if assembly_name in self.assemblies:
            for obj in self.assemblies[assembly_name]["objects"]:
                obj.metadata["assembly"] = None
            del self.assemblies[assembly_name]
            self._undo_push()
            self._notify("💥 Esploso")
    
    def import_file(self, file_path):
        try:
            mesh = trimesh.load(file_path, force='mesh')
            if isinstance(mesh, trimesh.Scene):
                mesh = mesh.dump(concatenate=True)
            
            if hasattr(mesh, 'extents') and np.any(np.array(mesh.extents) < 1):
                mesh.apply_scale(1000.0)
            
            mesh = validate_and_place_mesh(mesh)
            mesh.metadata.update({
                "layer": self.active_layer,
                "color": PASTEL_COLORS[self.color_idx % len(PASTEL_COLORS)],
                "name": Path(file_path).stem,
                "shape_type": "imported",
                "params": {},
                "assembly": None
            })
            
            self.color_idx += 1
            self.objects.append(mesh)
            self._undo_push()
            self._notify(f"✅ Importato: {Path(file_path).name}")
            return mesh
        except Exception as e:
            self._notify(f"❌ Errore import: {e}")
            return None
    
    def export_multi(self, file_path, format):
        try:
            visible_objects = [
                obj for obj in self.objects 
                if self.layers.get(obj.metadata.get("layer", "Default"), {}).get("visible", True)
            ]
            
            if not visible_objects:
                return False, "Vuoto"
            
            trimesh.Scene(visible_objects).export(file_path, file_type=format.lower())
            self._notify(f"✅ Esportato {format}")
            return True, f"✅ {format}"
        except Exception as e:
            print(f"⚠️ export_multi error: {e}")
            return False, "❌ Errore export"
    
    def boolean_op(self, operation):
        if len(self.objects) < 2 or not self.selected:
            return self._notify("⚠️ Servono almeno 2 oggetti")
        
        try:
            other_objects = [obj for obj in self.objects if obj is not self.selected]
            result = boolean_safe(self.selected, other_objects[0], operation)
            
            if result.is_empty:
                return self._notify("❌ Operazione risultante vuota")
                
            result.metadata.update(self.selected.metadata)
            result.metadata["name"] += f"_{operation}"
            
            self.objects = [result]
            self.selected = result
            self._undo_push()
            self._notify(f"✅ {operation} eseguita")
            return True
        except Exception as e:
            self._notify(f"❌ Boolean fallita: {e}")
            return False
    
    def shell(self, thickness):
        return self.boolean_op("Sottrazione") if len(self.objects) >= 2 else self._notify("⚠️ Servono 2 oggetti per il guscio")
    
    def _notify(self, message):
        print(message)

class CAMEngine:
    def __init__(self):
        self.toolpaths = []
    
    def generate_adaptive_3d(self, mesh, tool_diameter=3.0, stepover=0.5, feed_rate=1000, clearance=5.0, max_z_delta=2.0):
        self.toolpaths = []
        
        if not mesh or len(mesh.faces) == 0:
            return False
        
        try:
            min_bounds, max_bounds = mesh.bounds
            safe_z = max_bounds[2] + clearance
            y_current = min_bounds[1]
            direction_x = 1
            
            while y_current <= max_bounds[1]:
                x_points = np.arange(min_bounds[0], max_bounds[0] + stepover, stepover)
                if direction_x == -1:
                    x_points = x_points[::-1]
                
                origins = np.array([[x, y_current, safe_z] for x in x_points])
                vectors = np.tile([0, 0, -1], (len(x_points), 1))
                
                locations, indices, _ = mesh.ray.intersects_location(
                    origins, 
                    vectors, 
                    multiple_hits=False
                )
                
                path, previous_z = [], safe_z
                for i, x in enumerate(x_points):
                    hits = locations[indices == i]
                    target = hits[0][2] + tool_diameter / 2 if len(hits) > 0 else previous_z
                    
                    delta = target - previous_z
                    if abs(delta) > max_z_delta:
                        target = previous_z + np.sign(delta) * max_z_delta
                    
                    previous_z = target
                    path.append([x, y_current, previous_z])
                
                if len(path) > 1:
                    self.toolpaths.append({
                        "type": "ext", 
                        "pts": path, 
                        "s": feed_rate
                    })
                
                y_current += stepover
                direction_x *= -1
            
            return len(self.toolpaths) > 0
        except Exception as e:
            print(f"⚠️ generate_adaptive_3d error: {e}")
            return False
    
    def export_gcode_3d(self, file_path, clearance=5.0, rapid_rate=2000):
        if not self.toolpaths:
            return False
        
        try:
            lines = [
                f"; {APP_NAME} v{VERSION}",
                "G28; Home",
                "G90; Absolute",
                f"G1 Z{clearance} F{rapid_rate}"
            ]
            
            for toolpath in self.toolpaths:
                points, feed = toolpath["pts"], toolpath.get("s", 1000)
                lines.append(f"G1 X{points[0][0]:.3f} Y{points[0][1]:.3f} Z{points[0][2]:.3f} F{feed}")
                for point in points[1:]:
                    lines.append(f"G1 X{point[0]:.3f} Y{point[1]:.3f} Z{point[2]:.3f}")
            
            with open(file_path, "w", encoding="utf-8") as file:
                file.write("\n".join(lines))
            
            return True
        except Exception as e:
            print(f"⚠️ export_gcode_3d error: {e}")
            return False

    """
    
    def __init__(self, scene, window):
        """Inizializzazione del widget OpenGL"""
        super().__init__()
        self.scene = scene
        self.window = window
        self.rotation = [-35, -35]
        self.rotation_z = 0.0
        self.distance = 150.0
        self.pan = [0, 0, 0]
        self.interaction_mode = 'NONE'
        self.last_position = None
        self.active_handle = None
        self.drag_start_mouse = np.array([0, 0])
        self.drag_offset = np.array([0, 0, 0])
        self.setFocusPolicy(Qt.StrongFocus)
        self.setMouseTracking(True)
        self.sketch_mode = False
        self.sketch_tool = "line"
        self.sketch_points = []
        self.cam_mode = False
        self.angle_points = []
        self.fps_timer = QTimer()
        self.fps_timer.timeout.connect(self._update_stats)
        self.fps_timer.start(500)
        self.frames = 0
        self.last_time = time.time()
        self.device_pixel_ratio = 1.0
        self.physical_width = 1000
        self.physical_height = 800
        self.gizmo_points_3d = {}
        self.gizmo_points_2d = {}
        self._gl_ready = False
        self._modelview_matrix = None
        self._projection_matrix = None
        self._viewport = None
    
    def initializeGL(self):
        """Inizializzazione contesto OpenGL"""
        try:
            # Impostazione colore di sfondo
            glClearColor(0.06, 0.06, 0.08, 1.0)
            glClearDepth(1.0)
            
            # Configurazione depth buffer
            glEnable(GL_DEPTH_TEST)
            glDepthFunc(GL_LEQUAL)
            glDepthMask(GL_TRUE)
            
            # Configurazione backface culling (fondamentale per nascondere lo scheletro)
            glEnable(GL_CULL_FACE)
            glCullFace(GL_BACK)
            glFrontFace(GL_CCW)
            
            # Disattivazione di elementi non necessari
            glDisable(GL_BLEND)
            glDisable(GL_ALPHA_TEST)
            
            # Configurazione shading
            glShadeModel(GL_SMOOTH)
            glHint(GL_PERSPECTIVE_CORRECTION_HINT, GL_NICEST)
            
            # Configurazione illuminazione
            glEnable(GL_LIGHTING)
            glEnable(GL_LIGHT0)
            glEnable(GL_LIGHT1)
            glEnable(GL_COLOR_MATERIAL)
            glColorMaterial(GL_FRONT_AND_BACK, GL_AMBIENT_AND_DIFFUSE)
            
            # Parametri luce 0
            glLightfv(GL_LIGHT0, GL_AMBIENT, [0.5, 0.5, 0.5, 1.0])
            glLightfv(GL_LIGHT0, GL_DIFFUSE, [0.8, 0.8, 0.8, 1.0])
            glLightfv(GL_LIGHT0, GL_SPECULAR, [0.3, 0.3, 0.3, 1.0])
            glLightfv(GL_LIGHT0, GL_POSITION, [300, 300, 400, 0.0])
            
            # Parametri luce 1
            glLightfv(GL_LIGHT1, GL_DIFFUSE, [0.3, 0.3, 0.4, 1.0])
            glLightfv(GL_LIGHT1, GL_POSITION, [-200, -150, 200, 0.0])
            
            # Parametri modello illuminazione
            glLightModelfv(GL_LIGHT_MODEL_AMBIENT, [0.2, 0.2, 0.2, 1.0])
            
            # Configurazione normali
            glEnable(GL_NORMALIZE)
            
            # Configurazione offset per wireframe
            glEnable(GL_POLYGON_OFFSET_LINE)
            glEnable(GL_POLYGON_OFFSET_FILL)
            
            # Flag di inizializzazione completata
            self._gl_ready = True
            self.update()
        except Exception as e:
            print(f"⚠️ initGL: {e}")
            self._gl_ready = False
    
    def resizeGL(self, width, height):
        """Gestione ridimensionamento widget"""
        try:
            # Gestione pixel ratio
            self.device_pixel_ratio = self.devicePixelRatioF()
            if self.device_pixel_ratio <= 0:
                self.device_pixel_ratio = 1.0
            
            # Controllo dimensioni valide
            if width <= 0 or height <= 0:
                return
            
            # Calcolo dimensioni fisiche
            physical_width = int(width * self.device_pixel_ratio)
            physical_height = int(height * self.device_pixel_ratio)
            
            # Controllo dimensioni fisiche valide
            if physical_width <= 0 or physical_height <= 0:
                return
            
            # Aggiornamento variabili
            self.physical_width = physical_width
            self.physical_height = physical_height
            
            # Impostazione viewport
            glViewport(0, 0, physical_width, physical_height)
        except Exception as e:
            print(f"⚠️ resizeGL: {e}")
    
    def paintGL(self):
        """Rendering della scena 3D"""
        if not self._gl_ready or not self.isValid():
            return
        
        try:
            # Controllo dimensioni widget
            width = self.width()
            height = self.height()
            if width <= 0 or height <= 0:
                return
            
            # Gestione pixel ratio
            self.device_pixel_ratio = self.devicePixelRatioF()
            if self.device_pixel_ratio <= 0:
                self.device_pixel_ratio = 1.0
            
            # Calcolo dimensioni fisiche
            physical_width = int(width * self.device_pixel_ratio)
            physical_height = int(height * self.device_pixel_ratio)
            self.physical_width = physical_width
            self.physical_height = physical_height
            
            # Impostazione viewport
            glViewport(0, 0, physical_width, physical_height)
            
            # Pulizia buffer
            glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT)
            
            # Configurazione stato OpenGL (standard CAD)
            glDisable(GL_BLEND)
            glDepthMask(GL_TRUE)
            glEnable(GL_DEPTH_TEST)
            glDepthFunc(GL_LEQUAL)
            
            # Sincronizzazione camera
            self._sync_camera(width, height)
            
            # Disegno griglia
            self._grid()
            
            # Disegno oggetti non selezionati
            for obj in self.scene.objects:
                if obj is self.scene.selected:
                    continue
                layer = obj.metadata.get("layer", "Default")
                if not self.scene.layers.get(layer, {}).get("visible", True):
                    continue
                self._draw_mesh(obj, False)
            
            # Disegno oggetto selezionato
            if self.scene.selected:
                layer = self.scene.selected.metadata.get("layer", "Default")
                if self.scene.layers.get(layer, {}).get("visible", True):
                    self._draw_mesh(self.scene.selected, True)
                    if not self.sketch_mode:
                        self._gizmo()
            
            # Disegno sketch se attivo
            if self.sketch_mode:
                self._sketch_draw()
            
            # Disegno righelli
            self._draw_rulers_qt()
            
            # Aggiornamento contatore frame
            self.frames += 1
        except Exception as e:
            print(f"⚠️ paintGL: {e}")
    
    def _sync_camera(self, width, height):
        """Sincronizzazione matrici di trasformazione"""
        try:
            # Impostazione matrice proiezione
            glMatrixMode(GL_PROJECTION)
            glLoadIdentity()
            
            # Configurazione proiezione in base alla modalità
            if self.sketch_mode:
                glOrtho(-width / 2, width / 2, -height / 2, height / 2, -5000, 5000)
            else:
                aspect_ratio = self.physical_width / self.physical_height if self.physical_height > 0 else 1
                gluPerspective(45, aspect_ratio, 0.5, 1000.0)
            
            # Impostazione matrice modelview
            glMatrixMode(GL_MODELVIEW)
            glLoadIdentity()
            
            # Configurazione camera
            if not self.sketch_mode:
                gluLookAt(0, 0, self.distance, 0, 0, 0, 0, 1, 0)
                glTranslatef(*self.pan)
                glRotatef(self.rotation[0], 1, 0, 0)
                glRotatef(self.rotation[1], 0, 1, 0)
                glRotatef(self.rotation_z, 0, 0, 1)
            
            # Salvataggio matrici per picking
            self._modelview_matrix = glGetDoublev(GL_MODELVIEW_MATRIX)
            self._projection_matrix = glGetDoublev(GL_PROJECTION_MATRIX)
            self._viewport = glGetIntegerv(GL_VIEWPORT)
        except Exception as e:
            print(f"⚠️ _sync_camera: {e}")
    
    def _draw_mesh(self, mesh, is_selected):
        """Rendering di una mesh"""
        try:
            # Controllo validità mesh
            if len(mesh.vertices) == 0:
                return
            
            # Preparazione dati
            vertices = np.ascontiguousarray(mesh.vertices.astype(np.float32))
            normals = np.ascontiguousarray(
                mesh.vertex_normals.astype(np.float32) 
                if mesh.vertex_normals is not None 
                else np.zeros_like(vertices)
            )
            faces = np.ascontiguousarray(mesh.faces.astype(np.uint32))
            
            # Estrazione colore
            color = mesh.metadata.get("color", PASTEL_COLORS[0])
            if len(color) == 3:
                color = [*color, 1.0]
            
            # Configurazione attributi vertici
            glEnableClientState(GL_VERTEX_ARRAY)
            glEnableClientState(GL_NORMAL_ARRAY)
            glVertexPointer(3, GL_FLOAT, 0, vertices)
            glNormalPointer(GL_FLOAT, 0, normals)
            
            # Isolamento stato OpenGL
            glPushAttrib(GL_ALL_ATTRIB_BITS)
            try:
                # Configurazione base
                glDisable(GL_BLEND)
                glDisable(GL_ALPHA_TEST)
                glDepthMask(GL_TRUE)
                glPolygonMode(GL_FRONT_AND_BACK, GL_FILL)  # Assicura che il rendering sia in FILL
                glEnable(GL_CULL_FACE)  # Riattiva il backface culling
                glCullFace(GL_BACK)
                glFrontFace(GL_CCW)
                
                # Configurazione illuminazione
                glEnable(GL_LIGHTING)
                glShadeModel(GL_SMOOTH)
                glColor4f(color[0], color[1], color[2], 1.0)
                
                # Disegno mesh
                glDrawElements(GL_TRIANGLES, faces.size, GL_UNSIGNED_INT, faces)
                
                # Disegno wireframe SOLO per selezione (risolve il problema dello scheletro)
                if is_selected:
                    glPolygonOffset(-1.5, -1.5)
                    glEnable(GL_POLYGON_OFFSET_LINE)
                    glDisable(GL_LIGHTING)
                    glPolygonMode(GL_FRONT_AND_BACK, GL_LINE)
                    glLineWidth(2.0)
                    glColor4f(color[0], color[1], color[2], 1.0)
                    glDrawElements(GL_TRIANGLES, faces.size, GL_UNSIGNED_INT, faces)
                    glDisable(GL_POLYGON_OFFSET_LINE)
            finally:
                # Ripristino stato
                glPopAttrib()
            
            # Disabilitazione attributi vertici
            glDisableClientState(GL_VERTEX_ARRAY)
            glDisableClientState(GL_NORMAL_ARRAY)
        except Exception as e:
            print(f"⚠️ _draw_mesh: {e}")
    
    def _grid(self):
        """Disegno della griglia 3D"""
        try:
            # Isolamento stato
            glPushAttrib(GL_ENABLE_BIT | GL_CURRENT_BIT | GL_LINE_BIT | GL_DEPTH_BUFFER_BIT)
            
            # Configurazione base
            glDisable(GL_LIGHTING)
            glDisable(GL_DEPTH_TEST)
            glLineWidth(1.0)
            
            # Estensione griglia
            extent = 1000
            
            # Disegno linee griglia
            glBegin(GL_LINES)
            for i in range(-extent, extent + 1, 1):
                # Determinazione colore in base alla posizione
                if i == 0:
                    color = [0.35, 0.35, 0.40]
                elif i % 50 == 0:
                    color = [0.25, 0.25, 0.30]
                elif i % 5 == 0:
                    color = [0.18, 0.18, 0.22]
                else:
                    color = [0.12, 0.12, 0.15]
                
                # Impostazione colore
                glColor3f(*color)
                
                # Disegno linee X e Y
                glVertex3f(i, -extent, 0)
                glVertex3f(i, extent, 0)
                glVertex3f(-extent, i, 0)
                glVertex3f(extent, i, 0)
            glEnd()
            
            # Disegno assi principali
            glLineWidth(2.0)
            glBegin(GL_LINES)
            
            # Asse X (rosso)
            glColor3f(0.60, 0.25, 0.25)
            glVertex3f(-extent, 0, 0)
            glVertex3f(extent, 0, 0)
            
            # Asse Y (verde)
            glColor3f(0.25, 0.60, 0.35)
            glVertex3f(0, -extent, 0)
            glVertex3f(0, extent, 0)
            
            # Asse Z (blu)
            glColor3f(0.25, 0.35, 0.70)
            glVertex3f(0, 0, -extent)
            glVertex3f(0, 0, extent)
            
            glEnd()
            
            # Ripristino stato
            glPopAttrib()
        except Exception as e:
            print(f"⚠️ _grid: {e}")
    
    def _gizmo(self):
        """Disegno del gizmo di trasformazione"""
        if self._modelview_matrix is None or self._projection_matrix is None or self._viewport is None:
            return
        
        try:
            # Controllo validità oggetto selezionato
            obj = self.scene.selected
            if not obj or obj.bounds is None or len(obj.bounds) < 2:
                return
            
            # Estrazione bounding box
            bounds = obj.bounds
            center_x = (bounds[0][0] + bounds[1][0]) / 2
            center_y = (bounds[0][1] + bounds[1][1]) / 2
            z_bottom = bounds[0][2]
            z_top = bounds[1][2]
            
            # Calcolo parametri gizmo
            scale = self.distance / 100
            gap = 2.5 * scale
            size = 0.8 * scale
            
            # Pulizia cache gizmo
            self.gizmo_points_3d = {}
            self.gizmo_points_2d = {}
            
            # Generazione punti di controllo XY
            for i, x in enumerate([bounds[0][0], center_x, bounds[1][0]]):
                for j, y in enumerate([bounds[0][1], center_y, bounds[1][1]]):
                    key = f"g{i}_{j}"
                    z_position = z_bottom + gap
                    self.gizmo_points_3d[key] = [x, y, z_position]
                    
                    try:
                        # Proiezione 3D->2D
                        screen_x, screen_y, _ = gluProject(
                            x, y, z_position,
                            self._modelview_matrix,
                            self._projection_matrix,
                            self._viewport
                        )
                        self.gizmo_points_2d[key] = np.array([screen_x, self._viewport[3] - screen_y])
                    except Exception as e:
                        print(f"⚠️ _gizmo projection: {e}")
                        continue
                    
                    # Determinazione colore
                    if i == 1 and j == 1:
                        color = [0.7, 0.7, 0.7]
                    elif i != 1:
                        color = [1.0, 0.3, 0.3]
                    else:
                        color = [0.3, 0.85, 0.3]
                    
                    # Disegno cubo di controllo
                    glPushAttrib(GL_ENABLE_BIT | GL_CURRENT_BIT)
                    glDisable(GL_LIGHTING)
                    glColor4f(*color, 0.95)
                    glPushMatrix()
                    glTranslatef(x, y, z_position)
                    s = size
                    glBegin(GL_QUADS)
                    for vx, vy, vz in [
                        (-s, -s, s), (s, -s, s), (s, s, s), (-s, s, s),
                        (-s, -s, -s), (-s, s, -s), (s, s, -s), (s, -s, -s)
                    ]:
                        glVertex3f(vx, vy, vz)
                    glEnd()
                    glPopMatrix()
                    glPopAttrib()
            
            # Generazione punto di controllo Z
            try:
                arrow_height = 10 * scale
                base = z_top + gap * 1.5
                tip = base + arrow_height
                
                # Proiezione 3D->2D
                screen_x, screen_y, _ = gluProject(
                    center_x, center_y, tip,
                    self._modelview_matrix,
                    self._projection_matrix,
                    self._viewport
                )
                
                # Aggiornamento cache
                self.gizmo_points_2d['h'] = np.array([screen_x, self._viewport[3] - screen_y])
                self.gizmo_points_3d['h'] = [center_x, center_y, base]
                
                # Disegno freccia Z
                glPushAttrib(GL_ENABLE_BIT | GL_CURRENT_BIT)
                glDisable(GL_LIGHTING)
                glColor4f(0.3, 0.6, 1, 0.95)
                glBegin(GL_LINES)
                glVertex3f(center_x, center_y, base)
                glVertex3f(center_x, center_y, tip)
                glEnd()
                glPopAttrib()
            except Exception as e:
                print(f"⚠️ _gizmo Z-handle: {e}")
        except Exception as e:
            print(f"⚠️ _gizmo: {e}")
    
    def _draw_rulers_qt(self):
        """Disegno righelli di misura con QPainter"""
        try:
            # Controllo dimensioni widget
            width = self.width()
            height = self.height()
            if width <= 0 or height <= 0:
                return
            
            # Creazione painter
            painter = QPainter(self)
            painter.setRenderHint(QPainter.Antialiasing)
            painter.setFont(QFont("Segoe UI", 8))
            painter.setPen(QColor(180, 190, 200))
            
            # Calcolo parametri righelli
            range_x = int(self.distance * 2.5)
            start_x = int(self.pan[0] - range_x / 2)
            range_y = int(range_x * (height / width))
            start_y = int(self.pan[1] - range_y / 2)
            
            # Disegno righello X
            for x in range(start_x - (start_x % 10), start_x + range_x, 10):
                screen_x = (x - start_x) / range_x * width
                if 0 <= screen_x <= width:
                    tick_height = 10 if x % 50 == 0 else 5
                    painter.drawLine(int(screen_x), height, int(screen_x), height - tick_height)
                    if x % 50 == 0:
                        painter.drawText(int(screen_x) + 3, height - 12, str(x))
            
            # Disegno righello Y
            for y in range(start_y - (start_y % 10), start_y + range_y, 10):
                screen_y = height - ((y - start_y) / range_y * height)
                if 0 <= screen_y <= height:
                    tick_width = 10 if y % 50 == 0 else 5
                    painter.drawLine(0, int(screen_y), tick_width, int(screen_y))
                    if y % 50 == 0:
                        painter.drawText(12, int(screen_y) + 4, str(y))
            
            # Pulizia risorse
            painter.end()
        except Exception as e:
            print(f"⚠️ _draw_rulers_qt: {e}")
    
    def _sketch_draw(self):
        """Disegno delle entità di sketch"""
        try:
            # Isolamento stato
            glPushAttrib(GL_ENABLE_BIT | GL_CURRENT_BIT | GL_LINE_BIT)
            
            # Configurazione base
            glDisable(GL_LIGHTING)
            glLineWidth(1.5)
            
            # Estrazione colore layer
            color = self.scene.layers.get(self.scene.active_layer, {}).get("color", [0.35, 0.75, 0.9, 1])[:3]
            glColor3f(*color)
            
            # Disegno entità
            for entity in self.scene.sketch_entities:
                if "data" in entity:
                    if entity["type"] == "line":
                        glBegin(GL_LINES)
                    else:
                        glBegin(GL_LINE_LOOP)
                    for point in entity["data"]:
                        glVertex3f(point[0], point[1], 0)
                    glEnd()
            
            # Disegno punti sketch in corso
            if self.sketch_points:
                glColor3f(0.0, 1.0, 0.5)
                glBegin(GL_LINES)
                for point in self.sketch_points:
                    glVertex3f(point[0], point[1], 0)
                glEnd()
            
            # Ripristino stato
            glLineWidth(1.0)
            glEnable(GL_LIGHTING)
            glPopAttrib()
        except Exception as e:
            print(f"⚠️ _sketch_draw: {e}")
    
    def _update_stats(self):
        """Aggiornamento statistiche di rendering"""
        now = time.time()
        elapsed = now - self.last_time
        fps = self.frames / elapsed if elapsed > 0 else 0
        
        # Notifica al window manager
        if hasattr(self.window, 'update_stats'):
            self.window.update_stats(fps, 0, 0)
        
        # Reset contatori
        self.frames = 0
        self.last_time = now
    
    def _screen_to_ray(self, x, y):
        """Converte coordinate schermo in raggio 3D per picking"""
        if self._modelview_matrix is None or self._projection_matrix is None or self._viewport is None:
            width, height = self.width(), self.height()
            if width > 0 and height > 0:
                self._sync_camera(width, height)
        
        if self._modelview_matrix is None or self._projection_matrix is None or self._viewport is None:
            return None, None
        
        try:
            # Calcolo coordinate OpenGL
            gl_x = x * self.device_pixel_ratio
            gl_y = self.physical_height - (y * self.device_pixel_ratio)
            
            # Conversione schermo->mondo
            near_point = gluUnProject(
                gl_x, gl_y, 0.0,
                self._modelview_matrix,
                self._projection_matrix,
                self._viewport
            )
            far_point = gluUnProject(
                gl_x, gl_y, 1.0,
                self._modelview_matrix,
                self._projection_matrix,
                self._viewport
            )
            
            # Calcolo direzione
            ray_origin = np.array(near_point)
            ray_direction = np.array(far_point) - ray_origin
            norm = np.linalg.norm(ray_direction)
            if norm > 1e-6:
                ray_direction /= norm
            
            return ray_origin, ray_direction
        except Exception as e:
            print(f"⚠️ _screen_to_ray: {e}")
            return None, None
    
    def _pick_object(self, position):
        """Identifica l'oggetto sotto il cursore"""
        try:
            if not self.scene.objects:
                return None
            
            # Conversione schermo->mondo
            ray_origin, ray_direction = self._screen_to_ray(position.x(), position.y())
            if ray_origin is None or ray_direction is None:
                return None
            
            # Inizializzazione ricerca
            best = None
            min_t = float('inf')
            epsilon = 1e-9
            
            # Ricerca tra gli oggetti
            for obj in self.scene.objects:
                layer = obj.metadata.get("layer", "Default")
                if not self.scene.layers.get(layer, {}).get("visible", True):
                    continue
                
                # Bounding box test
                bounds = getattr(obj, 'bounds', None)
                if bounds is not None and len(bounds) == 2:
                    min_bound = np.array(bounds[0])
                    max_bound = np.array(bounds[1])
                    
                    # Calcolo intersezione con bounding box
                    t1 = (min_bound - ray_origin) / (ray_direction + epsilon)
                    t2 = (max_bound - ray_origin) / (ray_direction + epsilon)
                    t_enter = np.max(np.minimum(t1, t2))
                    t_exit = np.min(np.maximum(t1, t2))
                    
                    # Verifica intersezione
                    if t_enter <= t_exit and t_exit >= 0 and t_enter < min_t:
                        min_t = t_enter
                        best = obj
                        continue
                
                # Sfera di bounding test
                if hasattr(obj, 'vertices') and len(obj.vertices) > 0:
                    vertices = np.asarray(obj.vertices)
                    centroid = np.mean(vertices, axis=0)
                    radius = np.max(np.linalg.norm(vertices - centroid, axis=1))
                    
                    # Calcolo intersezione con sfera
                    oc = ray_origin - centroid
                    b = np.dot(oc, ray_direction)
                    c = np.dot(oc, oc) - radius * radius
                    discriminant = b * b - c
                    
                    # Verifica intersezione
                    if discriminant >= 0:
                        t = -b - np.sqrt(discriminant)
                        if t >= 0 and t < min_t:
                            min_t = t
                            best = obj
            
            return best
        except Exception as e:
            print(f"⚠️ _pick_object: {e}")
            return None
    
    def _pick_handle(self, position):
        """Identifica l'handle del gizmo sotto il cursore"""
        try:
            if not self.gizmo_points_2d:
                return None
            
            # Calcolo coordinate mouse
            mouse_x = position.x() * self.device_pixel_ratio
            mouse_y = position.y() * self.device_pixel_ratio
            
            # Ricerca handle
            for key, (screen_x, screen_y) in self.gizmo_points_2d.items():
                distance = math.hypot(mouse_x - screen_x, mouse_y - screen_y)
                if distance < 15:
                    return key
            
            return None
        except Exception as e:
            print(f"⚠️ _pick_handle: {e}")
            return None
    
    def _intersect_ray_plane(self, ray_origin, ray_direction, z):
        """Calcola intersezione raggio-piano"""
        try:
            if abs(ray_direction[2]) > 1e-6:
                t = (z - ray_origin[2]) / ray_direction[2]
                if t >= 0:
                    return ray_origin + t * ray_direction
            return None
        except Exception as e:
            print(f"⚠️ _intersect_ray_plane: {e}")
            return None
    
    def _get_sketch_coords(self, position):
        """Ottiene coordinate 3D per sketch"""
        try:
            ray_origin, ray_direction = self._screen_to_ray(position.x(), position.y())
            if ray_origin is not None and ray_direction is not None and abs(ray_direction[2]) > 1e-6:
                t = (0.0 - ray_origin[2]) / ray_direction[2]
                point = ray_origin + t * ray_direction
                return point[0], point[1]
            return 0.0, 0.0
        except Exception as e:
            print(f"⚠️ _get_sketch_coords: {e}")
            return 0.0, 0.0
    
    def mousePressEvent(self, event):
        """Gestione pressione tasto mouse"""
        try:
            self.last_position = event.pos()
            
            # Rotazione Z con Shift
            if event.modifiers() & Qt.ShiftModifier and event.button() == Qt.LeftButton:
                self.interaction_mode = 'ROT_Z'
                return
            
            # Pan con tasto centrale
            if event.button() == Qt.MiddleButton:
                self.interaction_mode = 'PAN'
                return
            
            # Gestione click sinistro
            if event.button() == Qt.LeftButton:
                # Modalità sketch
                if self.sketch_mode:
                    x, y = self._get_sketch_coords(event.pos())
                    self.sketch_points.append([x, y, 0.0])
                    if len(self.sketch_points) >= 2:
                        self.scene.sketch_entities.append({
                            "type": "line", 
                            "data": self.sketch_points.copy()
                        })
                        self.sketch_points.clear()
                    self.update()
                    return
                
                # Selezione handle gizmo
                handle = self._pick_handle(event.pos())
                if handle and self.scene.selected:
                    self.interaction_mode = 'DRAG_HANDLE'
                    self.active_handle = handle
                    self.drag_start_mouse = np.array([event.pos().x(), event.pos().y()])
                    self.scene._undo_push()
                    self.window.update_ui()
                    self.update()
                    return
                
                # Selezione oggetto
                hit = self._pick_object(event.pos())
                if hit:
                    self.scene.selected = hit
                    self.interaction_mode = 'DRAG_OBJ'
                    ray_origin, ray_direction = self._screen_to_ray(event.pos().x(), event.pos().y())
                    target = self._intersect_ray_plane(ray_origin, ray_direction, 0.0)
                    self.drag_offset = (target - np.array(hit.centroid)) if target is not None else np.zeros(3)
                    self.scene._undo_push()
                    self.window.update_ui()
                    self.update()
                else:
                    self.scene.selected = None
                    self.interaction_mode = 'ORBIT'
                    self.window.update_ui()
                    self.update()
        except Exception as e:
            print(f"⚠️ mousePressEvent: {e}")
    
    def mouseMoveEvent(self, event):
        """Gestione movimento mouse"""
        try:
            if self.last_position is None:
                return
            
            # Calcolo delta
            dx = event.x() - self.last_position.x()
            dy = event.y() - self.last_position.y()
            
            # Gestione modalità
            if self.interaction_mode == 'ROT_Z':
                self.rotation_z += dx * 0.5
                self.update()
            elif self.interaction_mode == 'PAN':
                self.pan[0] += dx * 0.5
                self.pan[1] -= dy * 0.5
                self.update()
            elif self.interaction_mode == 'DRAG_HANDLE' and self.scene.selected:
                # Calcolo fattore di scala
                factor = 1.0 + dx * 0.01
                factor = max(0.1, min(10, factor))
                
                # Applicazione trasformazione
                if 'g' in self.active_handle:
                    i = int(self.active_handle[1])
                    j = int(self.active_handle[3])
                    scale = [1, 1, 1]
                    if i in [0, 2]: scale[0] = factor
                    if j in [0, 2]: scale[1] = factor
                    if i == 1 and j == 1: scale[0] = scale[1] = factor
                    self.scene.selected.apply_scale(scale)
                elif self.active_handle == 'h':
                    self.scene.selected.apply_scale([1, 1, factor])
                
                self.window.update_ui()
                self.update()
            elif self.interaction_mode == 'DRAG_OBJ' and self.scene.selected:
                # Posizionamento oggetto
                ray_origin, ray_direction = self._screen_to_ray(event.pos().x(), event.pos().y())
                target = self._intersect_ray_plane(ray_origin, ray_direction, 0.0)
                if target is not None:
                    delta = target - self.drag_offset - np.array(self.scene.selected.centroid)
                    self.scene.selected.apply_translation([delta[0], delta[1], 0.0])
                    self.window.update_ui()
                    self.update()
            elif self.interaction_mode == 'ORBIT':
                # Rotazione camera
                self.rotation[1] += dx * 0.5
                self.rotation[0] += dy * 0.5
                self.rotation[0] = max(-89, min(89, self.rotation[0]))
                self.update()
            
            # Aggiornamento ultima posizione
            self.last_position = event.pos()
        except Exception as e:
            print(f"⚠️ mouseMoveEvent: {e}")
    
    def mouseReleaseEvent(self, event):
        """Gestione rilascio tasto mouse"""
        self.interaction_mode = 'NONE'
        self.last_position = None
    
    def wheelEvent(self, event):
        """Gestione rotella mouse"""
        try:
            self.distance -= event.angleDelta().y() * 0.2
            self.distance = max(10, min(5000, self.distance))
            self.update()
        except Exception as e:
            print(f"⚠️ wheelEvent: {e}")
    
    def keyPressEvent(self, event):
        """Gestione pressione tasti"""
        try:
            # Esci da modalità sketch
            if event.key() == Qt.Key_Escape:
                if self.sketch_mode:
                    self.window._toggle_sketch_mode()
                elif self.scene.selected:
                    self.scene.selected = None
                    self.window.update_ui()
                    self.update()
            
            # Elimina oggetto selezionato
            elif event.key() == Qt.Key_Delete and self.scene.selected:
                self.scene.delete()
                self.window.update_ui()
                self.update()
        except Exception as e:
            print(f"⚠️ keyPressEvent: {e}")

    def update_ui(self, obj):
        """Aggiornamento interfaccia con le proprietà dell'oggetto"""
        try:
            # Pulizia widget precedenti
            for widget in self.old_widgets:
                if widget:
                    widget.deleteLater()
            self.old_widgets.clear()
            self.spinboxes.clear()
            
            # Controllo oggetto valido
            if not obj or len(getattr(obj, 'vertices', [])) < 3:
                self.placeholder.show()
                self.coordinates_group.hide()
                self.status_group.hide()
                return
            
            # Visualizzazione informazioni
            self.placeholder.hide()
            self.coordinates_group.show()
            self.status_group.show()
            
            # Calcolo dimensioni
            vertices = np.asarray(obj.vertices)
            center = (vertices.min(0) + vertices.max(0)) / 2
            extents = vertices.max(0) - vertices.min(0)
            
            # Aggiornamento etichette
            self.x_label.setText(f"X:{center[0]:.2f}")
            self.y_label.setText(f"Y:{center[1]:.2f}")
            self.z_label.setText(f"Z:{center[2]:.2f}")
            self.width_label.setText(f"L:{extents[0]:.2f}")
            self.height_label.setText(f"H:{extents[2]:.2f}")
            self.depth_label.setText(f"P:{extents[1]:.2f}")
            self.volume_label.setText(f"{obj.volume:.1f} mm³")
            self.area_label.setText(f"{obj.area:.1f} mm²")
            self.watertight_label.setText("✅ Si" if obj.is_watertight else "⚠️ No")
            
            # Aggiunta parametri modificabili
            if obj.metadata.get("shape_type") in SHAPE_LIBRARY:
                for key, value in obj.metadata.get("params", {}).items():
                    if key in ("tipo", "spessore"):
                        continue
                    
                    # Creazione spinbox
                    spinbox = QDoubleSpinBox()
                    spinbox.setRange(0, 9999)
                    spinbox.setValue(float(value) if isinstance(value, (int, float)) else 0)
                    spinbox.valueChanged.connect(lambda v, k=key: self._on_param_changed(k, v))
                    
                    # Aggiunta al layout
                    self.params_layout.addRow(key.replace("_", " ").title(), spinbox)
                    self.spinboxes[key] = spinbox
                    self.old_widgets.append(spinbox)
        except Exception as e:
            print(f"⚠️ update_ui: {e}")

    def _on_param_changed(self, key, value):
        """Gestione modifica parametro"""
        if self.window and self.window.scene.selected:
            self.window.scene.selected.metadata["params"][key] = value

    def _toggle_sketch_mode(self):

        self.gl_widget.sketch_mode = not self.gl_widget.sketch_mode
        self.sketch_button.setChecked(self.gl_widget.sketch_mode)
        self.gl_widget.update()
        self._show_message("📐 Sketch " + ("ON" if self.gl_widget.sketch_mode else "OFF"))
    
    def _toggle_snap(self):
        """Attiva/disattiva lo snap magnetico"""
        self.scene.magnetic_snap = self.snap_button.isChecked()
        self._show_message("🎯 Snap " + ("ON" if self.scene.magnetic_snap else "OFF"))
    
    def _cam_ui(self):
        """Mostra il pannello CAM"""
        self._show_message("🛠️ Pannello CAM attivo")
    
    def _set_sketch_tool(self, tool):
        """Imposta lo strumento di sketching"""
        self.gl_widget.sketch_mode = True
        self.gl_widget.sketch_tool = tool
        self.sketch_button.setChecked(True)
        self.gl_widget.update()
        self._show_message(f"🛠️ Strumento: {tool}")
    
    def _extrude_sketch(self):
        """Estrude lo sketch corrente"""
        height, ok = QInputDialog.getDouble(self, "Estrudi", "Altezza", 10, 0.1, 500)
        if ok and self.scene.sketch_entities:
            self.scene.add_shape(
                "box", 
                {"larghezza": 20, "altezza": height, "profondità": 20}, 
                "Estruso"
            )
            self.scene.sketch_entities.clear()
            self.gl_widget.update()
            self._show_message("📤 Estruso")
    
    def _clear_sketch(self):
        """Pulisce lo sketch corrente"""
        self.scene.sketch_entities.clear()
        self.gl_widget.sketch_points.clear()
        self.gl_widget.update()
        self._show_message("🗑️ Sketch pulito")
    
    def _reset(self):
        """Resetta la scena corrente"""
        self.scene = Scene()
        self.gl_widget.scene = self.scene
        self.scene.selected = None
        self.update_ui()
        self.gl_widget.update()
        self._refresh_assembly_tree()
        self._show_message("🔄 Reset")
    
    def _import(self):
        """Importa un file nella scena"""
        path, _ = QFileDialog.getOpenFileName(
            self, 
            "Importa", 
            "", 
            "*.stl *.obj *.ply *.3mf"
        )
        if path:
            self.scene.import_file(path)
            self.update_ui()
            self.gl_widget.update()
            self._refresh_assembly_tree()
    
    def _export_3d(self):
        """Esporta la scena in formato 3D"""
        path, _ = QFileDialog.getSaveFileName(
            self, 
            "Esporta", 
            "", 
            "*.stl *.obj *.ply"
        )
        if path:
            success, message = self.scene.export_multi(path, Path(path).suffix[1:])
            self._show_message(message)
    
    def _export_dxf(self):
        """Mostra messaggio di avviso per esportazione DXF"""
        self._show_message("⚠️ DXF richiede ezdxf. Usa STL/OBJ.")
    
    def _group(self):
        """Raggruppa l'oggetto selezionato"""
        if self.scene.selected:
            self.scene.selected.metadata["assembly"] = f"asm_{int(time.time())}"
            self._refresh_assembly_tree()
            self._show_message("🔗 Raggruppato")
    
    def _explode(self):
        """Esplode l'assembly selezionato"""
        if self.scene.selected:
            self.scene.explode_assembly(self.scene.selected.metadata.get("assembly"))
            self._refresh_assembly_tree()
            self._show_message("💥 Esploso")
    
    def _refresh_assembly_tree(self):
        """Aggiorna l'albero degli assembly"""
        self.assembly_tree.clear()
        root = QTreeWidgetItem(self.assembly_tree, ["Scena"])
        root.setExpanded(True)
        free_group = QTreeWidgetItem(root, ["Libere"])
        for obj in self.scene.objects:
            if not obj.metadata.get("assembly"):
                QTreeWidgetItem(free_group, [obj.metadata.get("name", "Obj")])
    
    def _scan_ui(self):
        """Carica una scansione 3D"""
        path, _ = QFileDialog.getOpenFileName(
            self, 
            "Scan", 
            "", 
            "*.stl *.obj *.ply"
        )
        if path:
            mesh = self.scanner_module.load_scan(path)
            if mesh:
                self.scene.objects.append(mesh)
                self.gl_widget.update()
                self._refresh_assembly_tree()
                self._show_message("✅ Scan caricato")
    
    def update_ui(self):
        """Aggiorna l'interfaccia utente"""
        self.properties_panel.update_ui(self.scene.selected)
        self._update_ruler()
    
    def _update_ruler(self):
        """Aggiorna il righello di misura"""
        obj = self.scene.selected
        if obj and hasattr(obj, 'vertices') and len(obj.vertices) > 0:
            vertices = np.asarray(obj.vertices)
            center = (vertices.min(0) + vertices.max(0)) / 2
            extents = vertices.max(0) - vertices.min(0)
            self.ruler_label.setText(
                f"📏 {extents[0]:.1f}x{extents[2]:.1f}x{extents[1]:.1f} | 📍 {center[0]:.1f},{center[1]:.1f},{center[2]:.1f}"
            )
            self.ruler_label.show()
        else:
            self.ruler_label.hide()
    
    def update_stats(self, fps, draw_time, process_time):
        """Aggiorna le statistiche di rendering"""
        if hasattr(self, 'status_bar'):
            self.status_bar.showMessage(
                f"⚡ {fps:.0f} FPS | 🎨 {len(self.scene.objects)} oggetti"
            )
    
    def _generate_cam(self):
        """Genera percorsi CAM per l'oggetto selezionato"""
        if not self.scene.selected:
            return self._show_message("⚠️ Seleziona oggetto")
        
        success = self.cam_engine.generate_adaptive_3d(
            self.scene.selected, 
            self.tool_diameter.value(), 
            self.stepover.value(), 
            self.feed_rate.value()
        )
        self.gl_widget.update()
        QApplication.processEvents()
        self._show_message("✅ CAM generato" if success else "⚠️ CAM fallito")
    
    def _cam_export_gcode(self):
        """Esporta i percorsi CAM in G-Code"""
        if not self.cam_engine.toolpaths:
            return self._show_message("⚠️ Genera prima CAM")
        
        path, _ = QFileDialog.getSaveFileName(
            self, 
            "Salva G-Code", 
            "", 
            "*.gcode *.nc"
        )
        if path and self.cam_engine.export_gcode_3d(path):
            self._show_message("✅ G-Code salvato")
        else:
            self._show_message("❌ Errore G-Code")
    
    def _cam_simulate(self):
        """Mostra la simulazione CAM"""
        self._show_message("🎬 Simulazione attiva nel viewport")
    
    def _add_shape(self, shape_name):
        """Aggiunge una forma geometrica alla scena"""
        if shape_name in SHAPE_LIBRARY:
            shape_data = SHAPE_LIBRARY[shape_name].copy()
            shape_data["params"].update({"tipo": "Solido", "spessore": 0})
            self.scene.selected = self.scene.add_shape(
                shape_data["type"], 
                shape_data["params"], 
                shape_name.replace(" ", "_")
            )
            self.update_ui()
            # Forza l'aggiornamento immediato del rendering
            self.gl_widget.update()
            # Processa immediatamente gli eventi per mostrare subito il risultato
            QApplication.processEvents()
            self._refresh_assembly_tree()
            self._show_message(f"✅ {shape_name}")
    
    def _show_message(self, message):
        """Mostra un messaggio nella barra di stato"""
        if hasattr(self, 'status_bar'):
            self.status_bar.showMessage(message)
